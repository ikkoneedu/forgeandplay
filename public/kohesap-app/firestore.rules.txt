rules_version = '2';

// KO Ekip (forgeandplay.com) — Firestore güvenlik kuralları
// İlke: varsayılan KAPALI. Yalnızca Google ile giriş yapmış, e-postası doğrulanmış ve
// admin onayı (ya da adminin açtığı süreli davet kodu) almış kullanıcılar veriye erişir.
service cloud.firestore {
  match /databases/{database}/documents {

    // Google hesabıyla giriş (anonim / e-posta-şifre girişleri hiçbir şeye erişemez)
    function google() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'google.com'
        && request.auth.token.email_verified == true;
    }
    // Admin: doğrulanmış Google e-postası listede
    function isAdmin() {
      return google() && request.auth.token.email in ['carkci.caner@gmail.com'];
    }
    // KO Ekip erişimi: admin ya da erişim kaydı olan (onaylanmış) kullanıcı
    function hasKoAccess() {
      return google()
        && (isAdmin() || exists(/databases/$(database)/documents/kohesapAccess/$(request.auth.uid)));
    }
    function inviteDoc() {
      return get(/databases/$(database)/documents/kohesapAdmin/invite).data;
    }

    // ---- Üyelik planı (Üye / VIP) — uygulamadaki plan.js ve sunucudaki koekip-plan.ts ile aynı ----
    // config/plan { on, signup, teamVip, features: { id: 'uye'|'vip'|'kapali' } }
    // Plan yoksa ya da kapalıysa (on != true) her şey bugünkü gibi: tüm özellikler üyelere açık, ekip kurmak yalnızca admin.
    function planPath() { return /databases/$(database)/documents/config/plan; }
    function planDoc() { return get(planPath()).data; }
    function planOn() { return exists(planPath()) && planDoc().get('on', false) == true; }
    function featDef() { return {'sesli': 'vip', 'gb': 'vip', 'pazar': 'vip', 'ilan': 'vip'}; }
    // plan açıkken özelliğin katmanı
    function tierOn(f) { return planDoc().get('features', {}).get(f, featDef().get(f, 'uye')); }
    function accPath(uid) { return /databases/$(database)/documents/kohesapAccess/$(uid); }
    // VIP: süresi dolmamış vipUntil (ms) ya da admin kaydı
    function accVip(uid) {
      return exists(accPath(uid))
        && (get(accPath(uid)).data.get('code', '') == 'admin' || get(accPath(uid)).data.get('vipUntil', 0) > request.time.toMillis());
    }
    function featOk(f) {
      return isAdmin()
        || (!planOn() && f != 'ekipkur')
        || (planOn() && tierOn(f) == 'uye')
        || (planOn() && tierOn(f) == 'vip' && accVip(request.auth.uid));
    }

    // ---- Eski Forge&Play site verileri: kapatıldı (yalnızca admin) ----
    match /games/{d=**} { allow read, write: if isAdmin(); }
    match /rooms/{d=**} { allow read, write: if isAdmin(); }
    match /tournaments/{d=**} { allow read, write: if isAdmin(); }
    match /users/{d=**} { allow read, write: if isAdmin(); }
    match /leaderboard/{d=**} { allow read, write: if isAdmin(); }

    // Merkezi ayarlar (veri kaynakları, yayınlanan event / PUS listesi): üyeler okur, admin yazar
    match /config/{docId} {
      // plan belgesi: giriş yapan herkes okur (kayıt modu ve fiyat)
      allow read: if hasKoAccess() || (google() && docId == 'plan');
      allow write: if isAdmin();
    }

    // Canlı pazar: üyeler okur; üye uygulaması (exe) okuduğu pazarı paylaşır
    match /kohesapMarket/{docId} {
      // canlı pazar ve site ilanları VIP olabilir (plan)
      allow read: if hasKoAccess()
        && ((docId == 'latest' && featOk('pazar'))
            || (docId.matches('ilan-.*') && featOk('ilan'))
            || (docId != 'latest' && !docId.matches('ilan-.*')));
      allow write: if hasKoAccess()
        && request.resource.data.keys().hasOnly(['json', 'count', 'at', 'time', 'by', 'app'])
        && request.resource.data.json is string
        && request.resource.data.json.size() < 1000000
        && request.resource.data.by == request.auth.uid;
    }

    // Kişisel defter: herkes yalnızca kendi verisini okur/yazar
    match /kohesap/{uid} {
      allow read, write: if hasKoAccess() && request.auth.uid == uid;
      allow read: if isAdmin();
      match /chars/{charId} {
        allow read, write: if hasKoAccess() && request.auth.uid == uid;
        allow read: if isAdmin();
      }
    }

    // Davet kodu ayarı: yalnızca admin görür ve değiştirir
    match /kohesapAdmin/{docId} {
      allow read, write: if isAdmin();
    }

    // Erişim kaydı.
    //  - Varsayılan: yalnızca ADMIN ONAYI ile oluşur.
    //  - Admin davet kodunu açarsa (enabled + süre), kullanıcı geçerli kodla kendi kaydını oluşturabilir.
    //  - Kullanıcı sonra yalnızca son görülme / ad / karakter özetini günceller.
    match /kohesapAccess/{uid} {
      allow get: if google() && (request.auth.uid == uid || isAdmin());
      allow list: if isAdmin();
      allow create: if google() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['email', 'name', 'photo', 'at', 'code'])
        && request.resource.data.email == request.auth.token.email
        && exists(/databases/$(database)/documents/kohesapAdmin/invite)
        && inviteDoc().enabled == true
        && request.time.toMillis() < inviteDoc().exp
        && request.resource.data.code == inviteDoc().code;
      //  - Açık kayıt: admin planda kaydı "açık" yaptıysa Google ile gelen herkes kendi kaydını oluşturur (VIP alanı olmadan).
      allow create: if google() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['email', 'name', 'photo', 'at', 'code'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.code == 'acik'
        && exists(planPath())
        && planDoc().get('signup', 'onay') == 'acik';
      // vipUntil / vipBy yalnızca admin yazar; kullanıcı yalnızca "VIP olmak istiyorum" (vipReq) bırakır
      allow update: if google() && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['seen', 'name', 'photo', 'email', 'chars', 'vipReq'])
        && request.resource.data.email == request.auth.token.email
        && (!('vipReq' in request.resource.data) || request.resource.data.vipReq is int)
        && (!('chars' in request.resource.data) || request.resource.data.chars.size() <= 8);
      allow create, update, delete: if isAdmin();
    }

    // Erişim isteği: Google hesabıyla giren onaysız kullanıcı yalnızca kendi isteğini bırakır
    match /kohesapRequests/{uid} {
      allow create, update: if google() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['email', 'name', 'photo', 'at'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.name is string && request.resource.data.name.size() <= 60
        && request.resource.data.photo is string && request.resource.data.photo.size() <= 500;
      allow read, delete: if isAdmin() || (google() && request.auth.uid == uid);
    }

    // Uygulama varlıkları (sınıf görselleri, zamanlayıcı durumu, ses kullanımı): üyeler okur, admin yazar
    match /kohesapAssets/{docId} {
      allow read: if hasKoAccess();
      allow write: if isAdmin();
    }

    // Bildirim ayarları ve cihaz abonelikleri: herkes yalnızca kendi belgesi.
    // Gönderimi sunucu yapar; kohesapPushState yalnızca sunucuya açıktır (kural yok = kapalı).
    match /kohesapPush/{uid} {
      allow read, delete: if google() && request.auth.uid == uid;
      allow create, update: if hasKoAccess() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['subs', 'ev', 'boss', 'online', 'lead', 'team', 'server', 'name', 'evs', 'bossAll', 'bossIds', 'at'])
        && (!('subs' in request.resource.data) || request.resource.data.subs.size() <= 10)
        && (!('evs' in request.resource.data) || request.resource.data.evs.size() <= 60);
    }

    // Ekipler (en fazla 8 kişi). Yalnızca onaylı kullanıcılar görür; yalnızca admin kurar.
    match /kohesapTeams/{code} {
      function me() { return request.auth.uid; }
      function onlyMyEntry() {
        return request.resource.data.members.diff(resource.data.members).affectedKeys().hasOnly([me()]);
      }
      allow get: if hasKoAccess();
      allow list: if false;
      // ekip kurmak: admin ya da plana göre izinli üye (VIP sistemi kapalıyken yalnızca admin)
      allow create: if (isAdmin() || (hasKoAccess() && featOk('ekipkur')))
        && code.size() == 8
        && request.resource.data.owner == me()
        && request.resource.data.memberUids == [me()]
        && request.resource.data.members.keys().hasOnly([me()]);
      allow update: if hasKoAccess() && (
        // lider
        resource.data.owner == me()
        // üye: yalnızca kendi özeti
        || (me() in resource.data.memberUids
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'updatedAt'])
            && onlyMyEntry())
        // katılma: onaylı kullanıcı kendini ekler, en fazla 8 kişi
        || (!(me() in resource.data.memberUids)
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'memberUids', 'updatedAt'])
            && request.resource.data.memberUids.size() == resource.data.memberUids.size() + 1
            && request.resource.data.memberUids.hasAll(resource.data.memberUids)
            && me() in request.resource.data.memberUids
            && request.resource.data.memberUids.size() <= 8
            && onlyMyEntry())
        // ayrılma: kendini çıkarır
        || (me() in resource.data.memberUids
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'memberUids', 'updatedAt'])
            && request.resource.data.memberUids.size() == resource.data.memberUids.size() - 1
            && resource.data.memberUids.hasAll(request.resource.data.memberUids)
            && !(me() in request.resource.data.memberUids)
            && onlyMyEntry())
      );
      allow delete: if hasKoAccess() && resource.data.owner == me();

      // Ekip üyesi (ve hâlâ onaylı)
      function isMember() {
        return hasKoAccess()
          && request.auth.uid in get(/databases/$(database)/documents/kohesapTeams/$(code)).data.memberUids;
      }
      function isOwner() {
        return get(/databases/$(database)/documents/kohesapTeams/$(code)).data.owner == request.auth.uid;
      }

      // Ekip yazışması: üyeler okur ve kendi adına yazar; mesaj değiştirilemez, sahibi ya da lider siler.
      match /chat/{msgId} {
        allow read: if isMember();
        allow create: if isMember()
          && request.resource.data.u == request.auth.uid
          && request.resource.data.t is string
          && request.resource.data.t.size() <= 600
          && request.resource.data.keys().hasOnly(['u', 'n', 'c', 'cls', 'nat', 'race', 'rt', 't', 'at', 'ts', 'bot', 'kind', 'r'])
          && (!('r' in request.resource.data) || (request.resource.data.r is map && request.resource.data.r.size() <= 3));
        allow delete: if isMember() && (resource.data.u == request.auth.uid || isOwner());
      }

      // Boss sayaçları (sunucu başına bir belge): üyeler okur ve günceller.
      match /boss/{srv} {
        allow read: if isMember();
        allow create, update: if isMember() && srv.size() <= 24
          && request.resource.data.keys().hasOnly(['b', 'restart', 'at', 'by'])
          && request.resource.data.by == request.auth.uid
          && request.resource.data.b is map
          && request.resource.data.b.size() <= 60;
        allow delete: if isMember() && isOwner();
      }

      // Ekip planı (etkinlik + katılım): üyeler okur ve plan açar; ayrıntıyı açan ya da lider değiştirir / siler,
      // her üye yalnızca kendi katılım cevabını (rsvp.{uid}) yazar.
      //   plans/{id} → { t, ev, at, note, by, bn, rsvp: { uid: { s: 'y'|'m'|'n', c, cn, cls, lv, at } }, cAt,
      //                  att: { uid: 1 } (yoklama), loot: [{ i, w, wn, m, r, at }] (drop dağıtımı) }
      match /plans/{pid} {
        function planShapeOk() {
          let d = request.resource.data;
          return d.keys().hasOnly(['t', 'ev', 'at', 'note', 'by', 'bn', 'rsvp', 'cAt', 'att', 'loot'])
            && (!('att' in d) || (d.att is map && d.att.size() <= 9))
            && (!('loot' in d) || (d.loot is list && d.loot.size() <= 40))
            && d.t is string && d.t.size() >= 1 && d.t.size() <= 60
            && d.at is int
            && (!('ev' in d) || (d.ev is string && d.ev.size() <= 24))
            && (!('note' in d) || (d.note is string && d.note.size() <= 200))
            && (!('bn' in d) || (d.bn is string && d.bn.size() <= 40))
            && d.rsvp is map && d.rsvp.size() <= 8;
        }
        function myRsvpOk() {
          let r = request.resource.data.rsvp;
          return !(request.auth.uid in r) || (r[request.auth.uid] is map && r[request.auth.uid].s in ['y', 'm', 'n']);
        }
        allow read: if isMember();
        allow create: if isMember() && pid.size() <= 24 && planShapeOk() && myRsvpOk()
          && request.resource.data.by == request.auth.uid
          && request.resource.data.rsvp.keys().hasOnly([request.auth.uid]);
        allow update: if isMember() && planShapeOk() && (
          // açan ya da lider: ayrıntıları değiştirir (katılım cevaplarına ve açana dokunmadan)
          ((resource.data.by == request.auth.uid || isOwner())
            && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['rsvp', 'by']))
          // her üye: yalnızca kendi cevabı
          || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['rsvp'])
            && request.resource.data.rsvp.diff(resource.data.rsvp).affectedKeys().hasOnly([request.auth.uid])
            && myRsvpOk())
        );
        allow delete: if isMember() && (resource.data.by == request.auth.uid || isOwner());
      }

      // Sesli oda VIP olabilir: kullanıcı VIP ise ya da (ekip VIP'i açıksa) ekip lideri VIP / admin ise
      function voiceOk() {
        return featOk('sesli')
          || (planOn() && tierOn('sesli') == 'vip' && planDoc().get('teamVip', true) == true
              && accVip(get(/databases/$(database)/documents/kohesapTeams/$(code)).data.owner));
      }

      // Sesli sohbet odası
      match /voice/{uid} {
        allow read: if isMember();
        allow create, update: if isMember() && request.auth.uid == uid && voiceOk();
        allow delete: if isMember() && (request.auth.uid == uid || isOwner());
        // Bağlantı mesajları: üye başka bir üyenin kutusuna yazar, kutu sahibi okur ve siler.
        match /inbox/{msgId} {
          allow create: if isMember() && voiceOk()
            && request.resource.data.from == request.auth.uid
            && request.resource.data.keys().hasOnly(['t', 'sdp', 'c', 'from', 'fs', 'ts', 'at'])
            && (!('sdp' in request.resource.data) || request.resource.data.sdp.size() < 20000);
          allow read, delete: if google() && request.auth.uid == uid;
        }
      }
    }
  }
}
