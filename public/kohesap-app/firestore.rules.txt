rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Admin allow-list (Google sign-in provides token.email)
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email in ['carkci.caner@gmail.com'];
    }

    // Community games: anyone can read & play; only admin can create/delete.
    // Anyone signed in may increment the play counter (plays only).
    match /games/{gameId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin()
        || (request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['plays']));
    }

    // Global config (e.g. hidden built-in games): public read, admin write.
    match /config/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // LFG rooms: anyone can read; signed-in users can create/join/leave.
    // (Guest/anonymous sign-in counts as signed in.)
    match /rooms/{roomId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;

      // Room chat messages: public read, signed-in create.
      match /messages/{msgId} {
        allow read: if true;
        allow create: if request.auth != null;
      }
    }

    // Tournaments: public read; admin creates/deletes; signed-in users join/leave.
    match /tournaments/{tId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if request.auth != null;

      // Match result reports: public read (evidence); each user writes only their own.
      match /reports/{uid} {
        allow read: if true;
        allow create, update: if request.auth != null && request.auth.uid == uid;
      }
    }

    // Wallets: a user reads/writes only their own doc; admin may pay out to any.
    // NOTE: real-money integrity requires server-side (Cloud Functions) checks.
    match /users/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || isAdmin());
      allow write: if request.auth != null && (request.auth.uid == uid || isAdmin());
    }

    // KO Hesap canlı pazar: Windows uygulaması okuduğu pazarı paylaşır, web herkese gösterir.
    match /kohesapMarket/{docId} {
      allow read: if true;
      allow write: if request.auth != null
        && request.resource.data.keys().hasOnly(['json', 'count', 'at', 'time', 'by', 'app'])
        && request.resource.data.json is string
        && request.resource.data.json.size() < 1000000
        && request.resource.data.by == request.auth.uid;
    }

    // KO Hesap (Knight Online defteri): her kullanıcı yalnızca kendi verisini
    // okur/yazar. Ana belge + en fazla 4 karakter belgesi.
    match /kohesap/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      match /chars/{charId} {
        allow read, write: if request.auth != null && request.auth.uid == uid;
      }
    }

    // KO Hesap sınıf görselleri: herkes okur, yalnızca admin yazar.
    match /kohesapAssets/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // KO Hesap ekipleri (en fazla 8 kişi). Kodu bilen giriş yapmış kullanıcı
    // tek belgeyi okuyabilir (katılmak için); ekipler listelenemez.
    // Üye yalnızca kendi özetini günceller; katılma/ayrılma yalnızca kendini
    // ekler/çıkarır; lider her şeyi yapabilir (ad değiştirme, üye çıkarma).
    match /kohesapTeams/{code} {
      function signedIn() { return request.auth != null; }
      function me() { return request.auth.uid; }
      function onlyMyEntry() {
        return request.resource.data.members.diff(resource.data.members).affectedKeys().hasOnly([me()]);
      }
      allow get: if signedIn();
      allow list: if false;
      allow create: if signedIn()
        && code.size() == 8
        && request.resource.data.owner == me()
        && request.resource.data.memberUids == [me()]
        && request.resource.data.members.keys().hasOnly([me()]);
      allow update: if signedIn() && (
        // lider
        resource.data.owner == me()
        // üye: yalnızca kendi özeti
        || (me() in resource.data.memberUids
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'updatedAt'])
            && onlyMyEntry())
        // katılma: kendini ekler, en fazla 8 kişi
        || (!(me() in resource.data.memberUids)
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'memberUids', 'updatedAt'])
            && request.resource.data.memberUids.size() == resource.data.memberUids.size() + 1
            && request.resource.data.memberUids.hasAll(resource.data.memberUids)
            && me() in request.resource.data.memberUids
            && request.resource.data.memberUids.size() <= 8
            && onlyMyEntry())
        // ayrılma: kendini çıkarır
        || (me() in resource.data.memberUids
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'memberUids', 'updatedAt'])
            && request.resource.data.memberUids.size() == resource.data.memberUids.size() - 1
            && resource.data.memberUids.hasAll(request.resource.data.memberUids)
            && !(me() in request.resource.data.memberUids)
            && onlyMyEntry())
      );
      allow delete: if signedIn() && resource.data.owner == me();

      // Sesli sohbet odası: yalnızca ekip üyeleri görür; herkes yalnızca kendi kaydını yazar.
      function isMember() {
        return request.auth != null
          && request.auth.uid in get(/databases/$(database)/documents/kohesapTeams/$(code)).data.memberUids;
      }
      // Ekip yazışması: üyeler okur ve kendi adına yazar (bot mesajları da gönderenin adına);
      // mesaj değiştirilemez, sahibi ya da lider silebilir.
      match /chat/{msgId} {
        allow read: if isMember();
        allow create: if isMember()
          && request.resource.data.u == request.auth.uid
          && request.resource.data.t is string
          && request.resource.data.t.size() <= 600
          && request.resource.data.keys().hasOnly(['u', 'n', 'c', 'cls', 'nat', 'race', 'rt', 't', 'at', 'ts', 'bot', 'kind']);
        allow delete: if isMember() && (resource.data.u == request.auth.uid
          || get(/databases/$(database)/documents/kohesapTeams/$(code)).data.owner == request.auth.uid);
      }

      match /voice/{uid} {
        allow read: if isMember();
        allow create, update: if isMember() && request.auth.uid == uid;
        allow delete: if isMember() && (request.auth.uid == uid
          || get(/databases/$(database)/documents/kohesapTeams/$(code)).data.owner == request.auth.uid);
        // Bağlantı mesajları: üye başka bir üyenin kutusuna yazar, kutu sahibi okur ve siler.
        match /inbox/{msgId} {
          allow create: if isMember()
            && request.resource.data.from == request.auth.uid
            && request.resource.data.keys().hasOnly(['t', 'sdp', 'c', 'from', 'fs', 'ts', 'at'])
            && (!('sdp' in request.resource.data) || request.resource.data.sdp.size() < 20000);
          allow read, delete: if request.auth != null && request.auth.uid == uid;
        }
      }
    }

    // Public leaderboard stats (name / wins / earnings); written by admin payouts.
    match /leaderboard/{uid} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
